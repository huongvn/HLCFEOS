#!/bin/bash
#
# Python BMS Engine Update Script
#
# This script updates the Python BMS Engine to the latest version from Git.
# It replaces the old Node-RED update script.
#
# Usage:
#   ./update-bms-engine.sh
#
# The script will:
#   1. Pull latest code from Git
#   2. Install/update Python dependencies
#   3. Restart the bms-engine service
#

set -e

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# Functions
log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# Check if running as root
if [ "$EUID" -eq 0 ]; then
    log_warn "Running as root. It's recommended to run as regular user."
fi

# Get script directory
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
BMS_ENGINE_DIR="$(dirname "$SCRIPT_DIR")/code/python_engine"

log_info "Python BMS Engine Update Script"
log_info "================================"
echo ""

# Check if directory exists
if [ ! -d "$BMS_ENGINE_DIR" ]; then
    log_error "Python BMS Engine directory not found: $BMS_ENGINE_DIR"
    exit 1
fi

cd "$BMS_ENGINE_DIR"

# Step 1: Pull latest code
log_info "Step 1: Pulling latest code from Git..."
echo ""

if ! git pull; then
    log_error "Failed to pull latest code"
    exit 1
fi

log_info "Code updated successfully"
echo ""

# Step 2: Install/update dependencies
log_info "Step 2: Installing/updating Python dependencies..."
echo ""

if [ -f "requirements.txt" ]; then
    if ! pip3 install -r requirements.txt; then
        log_error "Failed to install dependencies"
        exit 1
    fi
    log_info "Dependencies installed successfully"
else
    log_warn "requirements.txt not found, skipping dependency installation"
fi
echo ""

# Step 3: Check devices.yaml symlink
log_info "Step 3: Checking devices.yaml symlink..."
echo ""

if [ ! -L "devices.yaml" ] && [ ! -f "devices.yaml" ]; then
    log_warn "devices.yaml not found, creating symlink..."
    if [ -f "../lvgl_project/devices.yaml" ]; then
        ln -s ../lvgl_project/devices.yaml devices.yaml
        log_info "Symlink created"
    else
        log_warn "devices.yaml source not found at ../lvgl_project/devices.yaml"
    fi
else
    log_info "devices.yaml exists"
fi
